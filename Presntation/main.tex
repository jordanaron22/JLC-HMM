\section{Model Description} 

\begin{frame}{Joint Latent Class Model}
\begin{itemize}
    \item Assumes latent class structure entirely captures the relationship between longitudinal and survival data
    \item Mixture of hidden Markov models (MHMM) for the longitudinal component
    \item Cox model for the survival aspect
    \item Joint model can incorporate uncertainty in latent class prediction better than a standard two-stage model
\end{itemize}
\end{frame}

\section{Estimation} 




\begin{frame}{EM Algorithm}
\begin{itemize}
    \item Write likelihood as if latent variables are known using indicator functions
    \item Replace log likelihood with the expectation of the log likelihood conditional on the observed data
\end{itemize}
\end{frame}

%------------------------------------------------
\begin{frame}{EM Algorithm}
% Define Q(theta|theta) as the expected value of the log likelihood function of theta with respect to the current conditional distribution of S  given X and the current estimates of the parameters

% The EM or expectation-maximization algorithm is an approach for performing maximum likelihood estimation in the presence of latent variables. It does this by first estimating the values for the latent variables in the e step, then optimizing the model in the m step. these two steps are repeated until convergence.

\begin{itemize}
    \item E-step: $Q(\theta|\theta^{(r)}) = E_{\theta^{(r)}}[\ell| \text{observed data}]$
    \item M-step: $\theta^{(r+1)} = \text{arg max}_\theta \text{ } Q(\theta|\theta^{(r)}) $
\end{itemize}
\end{frame}

\begin{frame}{Notation}
\begin{itemize}
    \item Let \textbf{a} and \textbf{x} be the observed activity and light data
    \item Let t be the observed survival time and $\delta$ be the censored indicator
    \item Let \textbf{S} be the latent sleep/wake state
    \item Let \textbf{G} be the latent subgroup assignment
    \item Observed: \textbf{a}, \textbf{x}, t, and $\delta$
    \item Latent: \textbf{S} and \textbf{G}
\end{itemize}
\end{frame}

\begin{frame}{Sandwich Estimator}
\small
Let $\ell_i(\theta)$ denote participant $i$'s weighted log-likelihood contribution.

\[
\resizebox{0.98\linewidth}{!}{$\displaystyle
\widehat{\mathrm{Var}}(\widehat\theta)
=
\underbrace{
\left[
-\sum_i
\left.\frac{\partial^2\ell_i(\theta)}{\partial\theta\,\partial\theta^T}
\right|_{\widehat\theta}
\right]^{-1}
}_{\text{bread}}
\underbrace{
\widehat{\mathrm{Var}}_d\!\left\{
\sum_i
\left.\frac{\partial\ell_i(\theta)}{\partial\theta}
\right|_{\widehat\theta}
\right\}
}_{\text{meat}}
\underbrace{
\left[
-\sum_i
\left.\frac{\partial^2\ell_i(\theta)}{\partial\theta\,\partial\theta^T}
\right|_{\widehat\theta}
\right]^{-T}
}_{\text{bread}}
$}
\]

\begin{itemize}
    \item \textbf{Bread:} inverse curvature of the weighted likelihood; converts score changes into parameter changes
    \item \textbf{Meat:} sampling variability of the weighted score; NHANES strata and PSUs determine this term
\end{itemize}

\end{frame}

\begin{frame}{Oakes Identity}

    Recall that $Q(\theta'|\theta)$ is the expected complete-data log-likelihood given the observed data and current parameter estimates

\[
\mathcal I_{\mathrm{obs}}(\widehat\theta)=
-\left[
\frac{\partial^2 Q(\theta'|\theta)}{\partial\theta'\partial\theta'^{\!T}}
+
\frac{\partial^2 Q(\theta'|\theta)}{\partial\theta'\partial\theta^{T}}
\right]_{\theta'=\theta=\widehat\theta}
\]
\medskip
\begin{itemize}
    \item The first term is the curvature of the M-step objective
    \item The second term accounts for changes in the E-step state and class probabilities
    \item Inverting $\mathcal I_{\mathrm{obs}}$ gives the bread for the sandwich estimator
\end{itemize}

\end{frame}


\begin{frame}{Design-Based Variance}
\small
Within PSU $j$ in stratum $h$, sum the weighted scores:
\[
\mathbf U_{hj}=\sum_{i\in(h,j)}\frac{\partial\ell_i(\widehat\theta)}{\partial\theta},
\qquad
\overline{\mathbf U}_h=\frac{1}{m_h}\sum_{j=1}^{m_h}\mathbf U_{hj}.
\]

The meat is the within-stratum variation of the PSU totals:
\[
\sum_h\frac{m_h}{m_h-1}\sum_{j=1}^{m_h}
(\mathbf U_{hj}-\overline{\mathbf U}_h)
(\mathbf U_{hj}-\overline{\mathbf U}_h)^T.
\]

where $m_h$ is the number of PSUs in stratum $h$.
\end{frame}

\begin{frame}{Murphy-Topel}
\small
\begin{center}
\resizebox{0.96\linewidth}{!}{%
\begin{tikzpicture}[>=stealth, node distance=1.05cm]
\node[draw, rounded corners, minimum width=2.25cm, minimum height=0.8cm] (hmm) {Longitudinal Model};
\node[draw, rounded corners, right=of hmm, minimum width=2.45cm, minimum height=0.8cm] (class) {Class probabilities};
\node[draw, rounded corners, right=of class, minimum width=2.3cm, minimum height=0.8cm] (cox) {Survival Model};
\draw[->, thick] (hmm) -- node[above] {} (class);
\draw[->, thick] (class) -- node[above] {} (cox);
\end{tikzpicture}
}
\end{center}

\begin{itemize}
    \item A naive Cox standard error treats class probabilities as fixed
    \item Murphy-Topel accounts for uncertainty introduced in stage 1 by formulating stacked estimating equations for both stages
    \item Full two-stage variance estimation uses Oakes information for the first-stage MHMM within a Murphy--Topel sandwich that propagates first-stage uncertainty into the survival estimates
\end{itemize}

\[
\underbrace{\sum_i w_i
\begin{bmatrix}
\mathbf u_{1i}(\widehat\theta_L)\\
\mathbf u_{2i}(\widehat\beta,\widehat\theta_L)
\end{bmatrix}}_{\mathbf U(\widehat\theta_L,\widehat\beta)}=\mathbf 0
\]
\[
\begin{gathered}
\theta_L=\text{longitudinal parameters}, \qquad
\beta=\text{survival parameters},\\
\mathbf u_{1i}=\text{Longitudinal score},\qquad
\mathbf u_{2i}=\text{Survival score}
\end{gathered}
\]




\end{frame}

\begin{itemize}
    \item Two-stage model estimates longitudinal parameters first, then plugs latent class prediction into Cox model
    \item Joint model reduces possible misclassification by allowing for uncertainty in latent class
    \item Incorporates $ P(G=g|\textbf{a,x},t)$ directly into Cox model
\end{itemize}
\end{frame}
